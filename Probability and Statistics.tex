\chapter{Introduction to Probability}
	\section{Probability Spaces}
		\begin{definition}[Probability Space]\index{probability space}
			A probability space is a measure space such that the measure of the entire space is equal to one. We denote the space $(\Omega, \mathcal{F}, P)$ or alternatively $(S, E, P)$ where
			\begin{itemize}
				\item $\Omega$ is an arbitrary non-empty set called the \emph{sample space}\index{sample space}. 
				\item $\mathcal{F}$ is the $\sigma$-algebra called the \emph{event space}\index{event space} where elements of $\mathcal{F}$ are called \emph{events}\index{events}. 
				\item The measure $P:\mathcal{F} \to [0, 1]$ is called the \emph{probability}\index{probability}. 
			\end{itemize}
		\end{definition}

	\section{Axioms of Probability}	
		\begin{definition}[Axioms of Probability]\index{axioms of probability}
			There are three axioms of probability which hold true in any probability space
			\begin{axiom}
				For every event $A\in \mathcal{F}$, $P(A) \geq 0$. 		
			\end{axiom} 
			\begin{axiom}
				For every sample space $\Omega$, $P(\Omega)=1$. 
			\end{axiom}
			\begin{axiom}
				For every infinite sequence of disjoint events $A_1, A_2, \dots$, 
				\begin{alignat}{4}
					P\left(\bigcup_{i=1}^\infty A_i\right)=\sum_{i=1}^\infty P(A_i)
				\end{alignat}
			\end{axiom}
		\end{definition}

		\begin{remark}[Measure Space]\index{measure space}
			The definition of a probability space and the axioms of probability imply that the probability space is also a measure space with a measure $P$ on $\mathcal{F}$ and thus the theory of measure spaces apply (which is the point of the definition). 
			
			Note that in the definition of a measure space if at least one set has finite measure then the requirement that $P(\emptyset)=0$ is implied and that by the definition of a probability space all measures are finite. 
		\end{remark}

	\section{Useful Corollaries}
	
		\begin{theorem}
			$P(\emptyset)=0$		
		\end{theorem}
		\begin{theorem}
			For every finite sequence of $n$ disjoint events $A_1, A_2, \dots, A_n$, 
				\begin{alignat}{4}
					P\left(\bigcup_{i=1}^n A_i\right)=\sum_{i=1}^n P(A_i)
				\end{alignat}
		\end{theorem}
		\begin{theorem}
			For every event $A$, $P(A^c)=1-P(A)$. 
		\end{theorem}
		
		\begin{theorem}
			If $A\subset B$ then $P(A)\leq P(B)$. 
		\end{theorem}
		
		\begin{theorem}
			For every event $A$, $0\leq P(A)\leq 1$. 
		\end{theorem}
		
		\begin{theorem}
			For every two events $A$ and $B$, 
			\begin{alignat}{4}
				P(A\cap B^c)=P(A)-P(A\cap B)
			\end{alignat}
		\end{theorem}
		
		\begin{theorem}\label{Theorem:1:1}
			For every two events $A$ and $B$, 
			\begin{alignat}{4}
				P(A \cup B) = P(A)+P(B)-P(A\cap B)
			\end{alignat}
		\end{theorem}
		
		\begin{remark}
			Theorem \ref{Theorem:1:1} says that the probability of a union of events is equal to their sum minus their intersection. 
		\end{remark}
		
		\begin{theorem}[Bonferroni Inequality]\index{Bonferroni inequality}
			For all events $A_1, A_2, \dots, A_n$, 
			\begin{alignat}{4}
				P\left(\bigcup_{i=1}^n A_i\right) \leq \sum_{i=1}^nP(A_i)
			\end{alignat}
			and
			\begin{alignat}{4}
				P\left(\bigcap{i=1}^n A_i\right) \geq 1-\sum_{i=1}^nP(A_i^c)
			\end{alignat}
		\end{theorem}
		
	\section{Set Theory}
	
		\begin{definition}[Mutually Exclusive]\index{mutually exclusive}
			Let $A$ and $B$ be events; we say that $A$ and $B$ are \emph{mutually exclusive} if and only if $A\cap B=\emptyset$. The family of sets $\set{A_\alpha}_{\alpha \in I}$ indexed by an arbitrary index set $I$ (regardless of cardinality) is mutually exclusive whenever $A_\alpha \cap A_\beta = \emptyset$ for all $\alpha \neq \beta$.  
		\end{definition}
		
\chapter{Conditional Probabilities}
	\section{Definition of Conditional Probability}
	
		\begin{definition}[Conditional Probability]\index{conditional probability}
			Suppose we know that some event $B$ occurred and we wish to know the probability that some other event $A$ has occurred; further assume that $P(B)>0$, then we define the \emph{conditional probability} of $A$ given $B$ as 
			\begin{alignat}{4}
				P(A\mid B) = \frac{P(A\cap B)}{P(B)}
			\end{alignat}
			NB: If $P(B)=0$ then $P(A\mid B)$ is undefined. 
		\end{definition}
		
		\begin{theorem}
			Let $A$ and $B$ be events and $P(B)>0$, then
			\begin{alignat}{4}
				P(A\cap B)=P(B)P(A\mid B)
			\end{alignat}
			and if $P(A)>0$, then 
			\begin{alignat}{4}
				P(A\cap B)=P(A)P(B\mid A)
			\end{alignat}
		\end{theorem}
	\section{Partitions}
	

		\begin{definition}[Partition]\index{partition}
			Let $\Omega$ be a sample space and let $A_1, A_2, \dots, A_k$ be a set of $k$ events such that $A_i \cap A_j = \emptyset$ for all $i$, $j$ and that $\bigcup_{i=1}^k A_i = \Omega$; then we say that $\left\{A_i\right\}_{i=1}^k$ form a \emph{partition} of $\Omega$
		\end{definition}
		
		\begin{theorem}[Law of total probability]\index{law of total probability}
			Suppose that $\set{B_i}_{i=1}^k$ partition $\Omega$ and that $P(B_i)>0$ for all $0<i<k$; then for every event $A$
			\begin{alignat}{4}
				P(A)=\sum_{i=1}^kP(B_i)P(A\mid B_i)
			\end{alignat}
		\end{theorem}

	\section{Independent Events}
	
		\begin{definition}[Independent Event]\index{independent events}
			Two events, $A$ and $B$ are \emph{independent} if $P(A\cap B) = P(A)P(B)$. 

			We say that a finite collection of events $\set{A_k}_{k=1}^n$ are independent if, for every sub-collection $\set{A_{i_k}}_{k=1}^j$ of $j$ of these events $(j=2,~3,~, \dots, k)$,
			\begin{alignat}{4}
				P\left(\bigcap_{k=1}^j A_{i_k}\right) = \prod_{k=1}^j P(A_{i_k})
			\end{alignat}

		\end{definition}
		
		\begin{theorem}
			If two events $A$ and $B$ are independent then $A$ and $B^c$ are also independent. 
		\end{theorem}

		\begin{theorem}
			Let $n>1$ and $\set{A_k}_{k=1}^n$ be events that are mutually exclusive. The events are also mutually independent if and only if all the events except possibly one of them has probability $0$. 
		\end{theorem}
		
	
		
		\begin{definition}[Conditional Independence]\index{conditional independence}\index{independence!conditional}
			We say that a finite collection of events $\set{A_k}_{k=1}^n$ are conditionally independent given $B$ if for every subcollection $\set{A_{i_k}}_{k=1}^j$ of $j$ of these events $(j=2,~3,~, \dots, k)$,
			\begin{alignat}{4}
				P\left(\bigcap_{k=1}^j A_{i_k}\mid B\right) = \prod_{k=1}^j P(A_{i_k}\mid B)
			\end{alignat}
		\end{definition}
		
		\begin{theorem}
			Suppose that $A_1$, $A_2$ and $B$ are events with $P(A_1\cap B)>0$, then $A_1$ and $A_2$ are conditionally independent given $B$ if and only if $P(A_2\mid A_1\cap B)=P(A_2\mid B)$. 
			\begin{proof}
				Suppose $A_1$ and $A_2$ are conditionally independent given $B$ then 
				\begin{alignat}{4}
					&P(A_1\cap A_2\mid B)&=P(A_1\mid B)P(A_2\mid B) \\
					\implies & P(A_1\cap A_2\mid B)&=\frac{P(A_1\cap B)}{P(B)}P(A_2\mid B)
				\end{alignat}
				by the definition of conditional probability on $P(A_1\mid B)$ noting that $P(A_1\cap B)>0\implies P(B)>0$. We can rearrange to write 
				\begin{alignat}{4}
					P(A_2\mid B)=\frac{P(A_1\cap A_2\mid B)P(B)}{P(A_1\cap B)}
				\end{alignat}
				Again using the definition of conditional probaiblity on $P(A_1\cap A_2\mid B)$ we write
				\begin{alignat}{4}
					&P(A_2\mid B)=\frac{P(A_1\cap A_2\cap B)P(B)}{P(A_1\cap B)P(B)} \\
					\implies & P(A_2\mid B)=\frac{P(A_1\cap A_2\cap B)}{P(A_1\cap B)} \label{Eqn:2:1}
				\end{alignat}
				Since intersection is associative we can rewrite $P(A_1\cap A_2\cap B)$ as $P(A_2 \cap (A_1\cap B))$ and using the definition of conditional probability write 
				\begin{alignat}{4}
					\frac{P(A_2 \cap (A_1\cap B))}{P(A_1\cap B)} &= P(A_2\mid A_1\cap B) \label{Eqn:2:2}
				\end{alignat}
				Combining \ref{Eqn:2:1} and \ref{Eqn:2:2} we conclude
				\begin{alignat}{4}
					P(A_2\mid B)&=P(A_2\mid A_1\cap B)
				\end{alignat}
				
				Conversely, suppose that $P(A_2\mid B)=P(A_2\mid A_1\cap B)$; we wish to show that $A_1$ and $A_2$ are conditionally independent given $B$; that is 
				\begin{alignat}{4}
					P(A_1\cap A_2\mid B)=P(A_1\mid B)P(A_2\mid B)
				\end{alignat}
				noting this implies 
				\begin{alignat}{4}
					P(A_2\mid B)=\frac{P(A_1\cap A_2\mid B)}{P(A_1\mid B)}
				\end{alignat}
				we make the previous argument in reverse and we compute
				\begin{alignat}{4}
					P(A_2\mid B)&= P(A_2\mid A_1\cap B)  \\
						&= \frac{P(A_1\cap A_2\cap B)}{P(A_1\cap B)} \\
						&=\frac{P(A_1\cap A_2\cap B)}{P(A_1\cap B)} \frac{P(A_1\mid B)}{P(A_1\mid B)} \\
						&=\frac{P(A_1\mid B)}{P(A_1\cap B)}\frac{P(A_1\cap A_2\cap B)}{P(A_1\mid B)} \\
						&=\frac{P(A_1\mid B)}{P(B)P(A_1\mid B)}\frac{P(A_1\cap A_2\cap B)}{P(A_1\mid B)} \\
						&=\frac{1}{P(B)}\frac{P(A_1\cap A_2\cap B)}{P(A_1\mid B)} \\
						&=\frac{1}{P(B)}\frac{P(B)P(A_1\cap A_2\mid B)}{P(A_1\mid B)} \\
						&=\frac{P(A_1\cap A_2\mid B)}{P(A_1\mid B)}
				\end{alignat}
				noting that we use $P(A\cap B)=P(B)P(A\mid B)$ as an alternate form of the definition of conditional probability where $A=A_1\cap A_2$ and that $P(A_1\cap B)>0$ guarantees that $P(A_1\cap B)$, $P(B)$ and $P(A_1\mid B)$ in the denominators are non-zero. 
			\end{proof}
		\end{theorem}
	
	\section{Bayes' Theorem}\index{Bayes' theorem}

		\begin{theorem}[Bayes' Theorem]\index{Bayes' theorem}
			Let the events $\set{B_j}_{j=1}^k$ form a partition of the space $\Omega$ such that $P(B_j)>0$ for $j=1,\dots, k$ and let $A$ be an event such that $P(A)>0$. Then, for $j=1, \dots, k$,
			\begin{alignat}{4}
				P(B_i\mid A)=\frac{P(B_i)P(A\mid B_i)}{\sum_{j=1}^kP(B_j)P(A\mid B_j)}
			\end{alignat}
		\end{theorem}

\chapter{Random Variables and Distributions}
	\section{Random Variables and Discrete Distributions}
		\begin{definition}[Random Variable]\index{random variable}
			Let $\Omega$ be a sample space; then a \emph{random variable} $X$ is a measurable function from $(\Omega, \mathcal{F})$ to $(S, \Sigma)$ where the latter is a measureable space. 
		\end{definition}
		
		\begin{remark}
			A random variable $X$ is most commonly a real number and the measureable space is $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$ where $\mathcal{B}(\mathbb{R})$ is the borel $\sigma$-algebra of $\mathbb{R}$. Many common properties of random variables cease to make sense if it isn't. One consequence is that singletons are measurable so it makes sense to talk about $X=5$. This need not be the case for more general random variables. 
		\end{remark}
		
		\begin{remark}
			That $X : (\Omega, \mathcal{F}, P) \to (S, \Sigma)$ is measurable implies that if $A\in\Sigma$ then $X^{-1}(A)\in\mathcal{F}$. This is important in the definition of distributions. 
		\end{remark}

		\begin{definition}
			Let $X : (\Omega, \mathcal{F}, P) \to (S, \Sigma)$ and let $A\in\Sigma$ then 
			\begin{alignat}{4}
				P(X\in A)=P(\set{\omega\in\Omega\mid X(\omega)\in A})
			\end{alignat}
			and 
			\begin{alignat}{4}
				P(X=x)=P(\set{\omega\in\Omega\mid X(\omega) = x}) \label{Eqn:3:1}
			\end{alignat}
			noting that equation \ref{Eqn:3:1} need not make sense when the random variable is not Borel-measurable. 
		\end{definition}
		
		\begin{definition}[Distribution]\index{distribution}
			Let $X$ be a random variable $X:(\Omega, \mathcal{F}, P)\to (S, \Sigma)$; the \emph{distribution} of $X$ is the probability measure $\mu_X$ on $(S, \Sigma)$ defined by $A\in\Sigma$, $\mu_X(A)=P(X^{-1}(A))$.  
		\end{definition}
		
		\begin{definition}[Discrete Distribution]\index{distribution!discrete}
			Let $X$ be a random variable, we say that $X$ has a discrete distribution if $X$ takes on a countable number of values. 
		\end{definition}
		
		\begin{definition}[Continuous Distribution]\index{distribution!continuous}
			Let $X$ be a random variable; if there exists a nonnegative function $f$, defined on the real line, such that for every interval of real numbers $[a,b]$,
			\begin{alignat}{4}
				P(a\leq X \leq b)=\int_a^b f(x)~dx
			\end{alignat}
			then $X$ is said to have a continuous distribution
		\end{definition}

		\begin{definition}[Probability Function]\index{probability function}\index{probability mass function}
			Let $X$ be a random variable with a discrete distribution; the probability function is the function $f$ defined by 
			\begin{alignat}{4}
				f(x)&=P(X=x) \\
					&=P(X^{-1}(\set{x}))
			\end{alignat}
		\end{definition}
		
		\begin{definition}[Support]\index{support}
			Let $X$ be a discrete random variable with distribution $f(x)=\mu_X(\set{x})=P(X=x)$, where $f$ is the distribution, then the support of $X$ is the closure of the set $\set{x\mid \mu_X(\set{x})>0}$. 
		\end{definition}

		\begin{definition}[Bernoulli Random Variable]\index{Bernoulli random variable}\index{random variable!Bernoulli}
			A random variable $Z$ that takes only two values; 0 and 1, with $P(Z=1)=p$ has the \emph{Bernoulli distribution} with parameter $p$. 
		\end{definition}
		
		\begin{definition}[Uniform Random Variable on Integers]\index{uniform random variable}\index{random variable!uniform}
			Let $a\leq b$ be integers and $X$ a random variable that is equally likely to take on each of the integers $a, \dots, b$; then we say that $X$ has the \emph{uniform distribution on the integers} $a, \dots, b$. 
		\end{definition}
		
		\begin{theorem}
			If $X$ has the uniform distribution on the integers $a, \dots, b$; then the probability function of $X$ is 
			\begin{alignat}{4}
				\mu_X(x)=\begin{cases}
					\frac{1}{b-a+1}, & x \in \{a, \dots, b\}, \\
					0, & \text{otherwise}.
				\end{cases}
			\end{alignat}
		\end{theorem}
		
		\begin{definition}[Binomial Random Variable]\index{random variable!binomial}
			A discrete random variable with distribution function given by 
			\begin{alignat}{4}
				\mu_X(x)=\begin{cases}
					\binom{n}{x} p^x(1-p)^{n-x} & \textrm{for $x=0, 1, \dots, n$}, \\
					0 & \textrm{otherwise}.
				\end{cases}
			\end{alignat}
			is called a \emph{binomial random variable} with parameters $n$ and $p$. 
		\end{definition}
		
		\begin{remark}
			The binomial random variable with parameters $n$ and $p$ arises when you have a sequence of $n$ Bernoulli random variables with probability $p$ of success and the random binomial random variable gives the probability of observing precisely $x$ successes. 
		\end{remark}

	\section{Continuous Random Variables}
	
		\subsection{The Probability Density Function}
			
			\begin{definition}[Continuous Random Variable]\index{random variable!continuous}
				We say that a random variable $X$ is a \emph{continuous random variable} if there exists a nonnegative function $f(x)$, defined on $\mathbb{R}$ such that for every interval of real numbers the probability that $X$ takes a value in the interval is
				\begin{alignat}{4}
					P(a\leq X \leq b)=\int_a^bf(x)dx
				\end{alignat}
			\end{definition}
			
			\begin{definition}[Probability Density Function]\index{probability density function}
				If $X$ is a continuous random variable the function $f$ in the definition of a continuous random variable is known as the \emph{probability density function}. 
			\end{definition}

	\section{The Cumulative Distribution Function}
	
		\begin{definition}[Distribution Function]\index{distribution function}\index{cumulative distribution function}
			The \emph{distribution function} or \emph{cumulative distribution function} of a real random variable $X$ is the function 
			\begin{alignat}{4}
				F(x)=P(X\leq x)
			\end{alignat}
		\end{definition}	

		\begin{theorem}\label{Theorem:3:1}
			A distribution function is non-decreasing
		\end{theorem}
		
		\begin{theorem}\label{Theorem:3:2}
			For a distribution function $F$, $\lim_{x\to-\infty}F(x)=0$ and $\lim_{x\to\infty}F(x)=1$
		\end{theorem}

		\begin{theorem}label{Theorem:3:3}
			Distribution functions are continuous from the right.
		\end{theorem}
		
		\begin{remark}
			Theorems \ref{Theorem:3:1} through \ref{Theorem:3:3} characterize CDFs; conversely any function for which these theorems holds is a distribution. 
		\end{remark}
		\begin{remark}
			A remarkable number of computations are made easy by the Fundamental Theorem of Calculus. 
		\end{remark}

		\begin{theorem}
			Let $X$ have a continuous distribution with density $f$ and distribution $F$, then $F$ is continuous and 
			\begin{alignat}{4}
				F(x)=\int_{-\infty}^xf(t)dt
			\end{alignat}
			and
			\begin{alignat}{4}
				\frac{dF(x)}{dx} = f(x)
			\end{alignat}
		\end{theorem}

		\begin{definition}[Quantiles]\index{quantiles}\index{Percentiles}
			Let $X$ be a random variable with CDF $F$. For each $p\in(0, 1)$, define $Q(p)$ to be the smallest value $x$ such that $F(x) \geq p$. This $Q(p)$ is called the $p$ \emph{quantile} of $X$ or the $100p$ \emph{percentile} of $X$. The function $Q:(0, 1)\to \mathbb{R}$ is the \emph{quantile function}\index{quantile function}. 
		\end{definition}
		
		\begin{definition}[Median]\index{median}
			The median of a random variable $X$ is the 0.5 quantile or 50th percentile. 
		\end{definition}
		
		\begin{remark}
			The median is special in that there is an alternative definition of the median. While the 50th percentile is the smallest value $x$ such that $F(x)\geq 0.5$ it may be that there's an interval of values such that $F(x)=0.5$. In that case some people take the middle of the interval to be the median. 
		\end{remark}
		
		
	\section{Bivariate Distributions}
		\begin{definition}[Joint Distribution]\index{joint distribution}\index{distribution!joint}
			Let $(\Omega, \mathcal{F}, P)$ be a probability space and let $X, Y$ be two real random variables on $\Omega$; then the map $(X, Y): \Omega \to \mathbb{R}^2$ is measurable. For every Borel set $C\in\mathcal{B}(\mathbb{R}^2)$ the preimage $\set{(X, Y)\in C} = \set{\omega\in\Omega\mid (X(\omega), Y(\omega))\in C)}\in\mathcal{F}$ is an event. 
			
			The \emph{joint distribution} of $X$ and $Y$ is the probability measure $\mu_{X, Y}$ on $(\mathbb{R}^2, \mathcal{B}(\mathbb{R}^2))$ defined by 
			\begin{alignat}{4}
				\mu_{X, Y}(C) = P(\set{(X, Y)\in C})
			\end{alignat}
			for $C\in\mathcal{B}(\mathbb{R}^2)$
		\end{definition}

		\begin{theorem}
			Suppose that two random variables $X$ and $Y$ on the same probability space each have a discrete distribution; then $X$ and $Y$ have a discrete joint distribution. 
		\end{theorem}

		\begin{definition}[Joint Probability Function]\index{joint probability function}
			The \emph{joint probability function} or the \emph{joint p.f.} of $X$ and $Y$ (defined on the same probability space) is defined as the function $f$ on the \emph{xy}-plane,
			\begin{alignat}{4}
				f(x, y)=P(X = x, ~Y = y)
			\end{alignat}
		\end{definition}


		\begin{definition}[Continuous Joint Distribution]\index{continuous joint distribution}\index{joint p.d.f}
			Two random variables $X$ and $Y$ (defined on the same probability space) have a \emph{continuous joint distribution} if there exists a nonnegative function $f$ defined over the $xy$-plane such that for every subset $C$ of the plane,
			\begin{alignat}{4}
				P[(X, Y)\in C] = \int\int_C f(x, y)~dx~dy
			\end{alignat}
			if the integral exists. The function $f$ is called the \emph{joint probability density function} of $X$ and $Y$. The closure of the set $\set{(x, y)\in\mathbb{R}^2\mid f(x, y)>0}$ is called the \emph{support} of $(X, Y)$. 
		\end{definition}
		
		\begin{theorem}
			A joint p.d.f. must satisfy the following two conditions:
			\begin{alignat}{4}
				f(x, y)\geq 0 \textrm{ for } x\in\mathbb{R} \textrm{ and } y\in\mathbb{R}
			\end{alignat}
			and
			\begin{alignat}{4}
				\int_{-\infty}^\infty \int_{-\infty}^\infty f(x, y)~dx~dy = 1
			\end{alignat}
		\end{theorem}
		
		\begin{theorem}
			For every continuous joint distribution on the $xy$-plane, the following two statements hold:
			\begin{enumerate}
				\item Every individual point, and every infinite sequence of points, in the $xy$-plane has probability $0$.
				\item Let $\gamma$ be a a continuous curve $\gamma: [s_1, s_2]\to \mathbb{R}^2$ of finite length then the image $\gamma([s_1, s_2])$ has probability zero; that is
				\begin{alignat}{4}
					P( (X, Y)\in\gamma([s_1, s_2])) = 0
				\end{alignat}
			\end{enumerate}
		\end{theorem}
		
		\begin{definition}[Joint Distribution Function]\index{joint distribution function}
			The \emph{joint distribution function} of two random variables $X$ and $Y$ is defined as the function $F$ such that for all values of $x$ and $y$
			\begin{alignat}{4}
				F(x, y) = P(X \leq x, Y \leq y)
			\end{alignat}		
		\end{definition}
		
		\begin{remark}
			If the joint cumulative distribution of two arbitrary random variables is $F$ then by the Fundamental Theorem of Calculus we can see that 
			\begin{alignat}{4}
				P(a < X \leq b, c < y \leq d) = F(b, d) - F(a, d) - F(b, c) + F(a, c)
			\end{alignat}
			This defines the distribution on rectangles from which we can construct more interesting sets.  
		\end{remark}

		\begin{theorem}
			Let $X$ and $Y$ have a joint distribution $F$; the distribution $F_1$ of a single random variable $X$ is
			\begin{alignat}{4}
				F_1(x)=\lim_{y\to\infty}F(x, y)
			\end{alignat}
		\end{theorem}

			\begin{problem}[\S 3.4 \# 5]
				Suppose that the joint distribution of two continuous random variables $X$ and $Y$ is as follows:
				\begin{alignat}{4}
					f(x, y) = \begin{cases}
						c \left(x^2 + y\right),& \textrm{for } 0\leq y \leq 1-x^2, \\
						0,&\textrm{otherwise}
					\end{cases}
				\end{alignat}
				Determine:
				\begin{enumerate}
					\item The value of the normalizing constant $c$.
					\item $P\left(0\leq X \leq \frac{1}{2}\right)$.
					\item $P(Y\leq X + 1)$.
					\item $P\left(Y=X^2\right)$.
				\end{enumerate}
				\begin{proof}
					We first consider the graph of $y=1-x^2$. This is a quadratic curve with vertex at $(0, 1)$. This intersects the $x$-axis where $y=0$ at $x=-1$ and $x=1$.  
					\begin{enumerate}
						\item We integrate the density over the area, which is the area above the $x$-axis and under the quadratic $1-x^2$ and note that it must be equal to 1
							\begin{alignat}{4}
								\int_{-1}^1\int_{0}^{1-x^2} c\left(x^2+y\right)~dy~dx & = c \int_{-1}^1 \left[ x^2y + \frac{y^2}{2}\right]_{y=0}^{y=1-x^2} ~ dx \\
									&= c\int_{-1}^1\left[ x^2(1-x^2)+\frac{\left(1-x^2\right)^2}{2} \right] ~ dx \\
									&=c\int_{-1}^1 \left(x^2 - x^4\right) + \frac{1}{2}\left(1-2x^2+x^4\right)~dx \\
									&=\frac{c}{2}\int_{-1}^1 2x^2-2x^4+1-2x^2+x^4 ~ dx \\
									&=\frac{c}{2} \int_{-1}^1 -x^4+1 ~ dx \\
									&=\frac{c}{2} \left[ -\frac{1}{5}x^5 + x\right]_{x=-1}^1 \\
									&=\frac{c}{2} \left[ 1-\frac{1}{5} +1 - \frac{1}{5}  \right] \\
									&=\frac{c}{2} \frac{8}{5} \\
									&={c}\frac{4}{5}
							\end{alignat}
							and so $c=\frac{5}{4}$. 
						\item To compute $P\left(0\leq X \leq \frac{1}{2}\right)$ we integrate the density over the bounds
							\begin{alignat}{4}
								\int_{0}^\frac{1}{2}\int_{0}^{1-x^2} \frac{5}{4}\left(x^2+y\right)~dy~dx &=\frac{5}{4} \int_{0}^\frac{1}{2} \left[ x^2y + \frac{y^2}{2}\right]_{y=0}^{y=1-x^2} ~ dx \\
									&= \frac{5}{4}\int_{0}^\frac{1}{2}\left[ x^2(1-x^2)+\frac{\left(1-x^2\right)^2}{2} \right] ~ dx \\
									&=\frac{5}{4}\int_{0}^\frac{1}{2}\ \left(x^2 - x^4\right) + \frac{1}{2}\left(1-2x^2+x^4\right)~dx \\
									&=\frac{5}{8}\int_{0}^\frac{1}{2}\ 2x^2-2x^4+1-2x^2+x^4 ~ dx \\
									&=\frac{5}{8} \int_{0}^\frac{1}{2}\ -x^4+1 ~ dx \\
									&=\frac{5}{8} \left[ -\frac{1}{5}x^5 + x\right]_{x=0}^{\frac{1}{2}} \\
									&=\frac{5}{8} \left[\frac{1}{2} - \frac{1}{5}\frac{1}{32} \right]\\
									&=\frac{1}{8}\left[\frac{5}{2}-\frac{1}{32}\right] \\
									&=\frac{5}{16}-\frac{1}{256} \\
									&=\frac{79}{256}
							\end{alignat}
						\item For $P(Y\leq X+1)$ we need to consider two areas, for $x<0$ we have $y=x+1$ as the upper bound and for $x\geq0$ we have $1-x^2$ as our upper bound. Again our density is $\frac{5}{4}\left(x^2+y\right)$. 
						\begin{alignat}{4}
							P(Y\leq X+1)&=\int_{-1}^0\int_{0}^{x+1}\frac{5}{4}\left(x^2+y\right)~dy~dx + \int_{0}^1\int_{0}^{1-x^2}\frac{5}{4}\left(x^2+y\right)~dy~dx
						\end{alignat}
						We compute each term separately 
						\begin{alignat}{4}
							\int_{-1}^0\int_{0}^{x+1}\frac{5}{4}\left(x^2+y\right)~dy~dx&=\int_{-1}^0 \frac{5}{4}\left[x^2y+\frac{1}{2}y^2\right]_{y=0}^{x+1} ~dx \\
								&=\frac{5}{4}\int_{-1}^0\left(x^2(x+1)+\frac{1}{2}\left(x+1\right)^2\right)~dx \\
								&=\frac{5}{8}\int_{-1}^0\left(2x^3+2x^2+x^2+2x+1\right)~dx \\
								&=\frac{5}{8}\int_{-1}^0\left(2x^3+3x^2+2x+1\right)~dx \\
								&=\frac{5}{8}\left[ \frac{2}{4}x^4 + x^3 + x^2+x \right]_{x=-1}^0 \\
								&=\frac{5}{8}\left(-\frac{1}{2} + 1-1 +1\right) \\
								&=\frac{5}{8}\frac{1}{2} \\
								&=\frac{5}{16}
						\end{alignat}
						and
						\begin{alignat}{4}
							\int_{0}^1\int_{0}^{1-x^2} \frac{5}{4}\left(x^2+y\right)~dy~dx &= \frac{5}{4} \int_{0}^1 \left[x^2y+\frac{1}{2}y^2\right]_{y=0}^{1-x^2} ~dx \\
								&=\frac{5}{8}\int_{0}^1\left[-x^4+1\right]~dx \\
								&=\frac{5}{8}\left[-\frac{1}{5}x^5 + x\right]_{x=0}^1 \\
								&=\frac{5}{8}\frac{4}{5} \\
								&=\frac{1}{2}
						\end{alignat}
						and so 
						\begin{alignat}{4}
							P(Y\leq X+1) = \frac{5}{16} + \frac{1}{2} &= \frac{13}{16}
						\end{alignat}
						\item Since the curve is 1d $P(Y=X^2) = 0$
					\end{enumerate}
				\end{proof}
			\end{problem}

	\section{Statistical Inference}
	
		\begin{definition}[Statistical Model]\index{statistical model}
			A \emph{statistical model} consists of
			\begin{itemize}
				\item an identification of random variables of interest (observable and hypothetically observable)
				\item a specification of a joint distribution or family of possible joint distributions for the observable random variables 
				\item the identification of any parameters of those distributions that are assumed unknown and possibly hypothetically observable
				\item and if desired, a specification for the (joint) distribution for the unknown parameter(s)
			\end{itemize}
			When we treat the unknown parameters $\theta$ as random, then the joint distribution of the observable random variables indexed by $\theta$ is understood as the conditional distribution of the observable random variables given $\theta$. 
		\end{definition}
		
		\begin{definition}[Statistical Inference]\index{statistical inference}
			A \emph{statistical inference} is a procedure that produces a probabilistic statement about some or all parts of a statistical model.
		\end{definition}
		
		\begin{definition}[Parameter]\index{parameter}
			In a problem of statistical inference, a characteristic or combination of characteristics that determine the joint distribution for the random variables of interest is called a parameter of the distribution. The set $\Theta$ of all possible values of a parameter $\theta$ or a vector of parameters $\set{\theta_i}_{i=1}^k$ is called the \emph{parameter space}. 
		\end{definition}

		\begin{remark}
			In a Bayesian setting where the unknown parameter $\theta$ is treated as a random variable we note that $\Omega$ is the sample space. We equip it with a $\sigma$-algebra $\mathcal{F}_{Omega}$, typically the Borel $\sigma$-algebra, and put a probability measure $\pi$ (the prior) on it; then $(\Omega, \mathcal{F}_\Omega, \pi)$ becomes a probability space and $\theta$ is a random variable on this space. 
		\end{remark}
		
		\begin{remark}[General Classes of Problems]
			There are several classes of problems:
			\begin{itemize} 				
				\item \emph{Prediction}: We try to predict the value of random variables that have not been observed. 
				\item \emph{Decision}: We try to make a decision based on the value of random variables that have not been observed. That decision will be with respect to a loss function defined as part of the problem.
				\item \emph{Experiment Design}: We make a decision on what experiment to conduct or the parameters of the experiment up to resource constraints. 
			\end{itemize}
		\end{remark}
		
		\begin{definition}[Statistic]\index{statistic}
			Suppose we have a set of \emph{observable} random variables $\set{X_i}_{i=1}^k$ then an arbitrary real valued function of $k$ real variables $T=r(X_1, \dots, X_k)$ is a statistic. 
		\end{definition}
		
		\begin{remark}
			NB: A statistic differs from a random variable as a function of $k$ random variable in that the it is a function of the data alone.
		\end{remark}
		
	\section{Prior and Posterior Distributions}
	
		\begin{definition}[Prior Distribution]\index{distribution!prior}\index{prior distribution}
			Suppose that we have a statistical model with a parameter $\theta$. We may treat $\theta$ as a random variable; then the distribution we assign to $\theta$ before observing other random variables of interesting in the model is called the \emph{prior distribution}
		\end{definition}
		
		\begin{remark}
			DeGroot and Schevrish commonly use $\xi(\theta)$ for the prior, other texts more commonly use $\pi(\theta)$ for the prior. In some contexts $p(\theta)$ is also used. 
		\end{remark}
		
		\begin{definition}[Posterior Distribution]\index{distribution!posterior}\index{posterior distribution}
			Consider a statistical inference problem with parameter $\theta$ and random variables $\set{X_\alpha}_{\alpha\in I}$ to be observed. The conditional distribution of $\theta$ given $\set{X_\alpha}_{\alpha\in I}$ is called the \emph{posterior distribution} of $\theta$ and is denoted $\xi(\theta\mid \set{X_\alpha}_{\alpha\in I})$
		\end{definition}
		
		\begin{remark}
			While there's a whole theory of posteriors with respect to large sets of observations we will typically deal with finite sets here $\set{X_i}_{i=1}^k$. There are issues with some of the following definitions and theorems in an infinite context we won't deal with for now. 
		\end{remark}
		
		\begin{theorem}
			Suppose that the $n$ random variables $\set{X_i}_{i=1}^n$ form a random sample from a distribution for which the density is $f(x\mid\theta)$. Suppose also that the value of the parameter $\theta$ is unknown and the prior density of $\theta$ is $\xi(\theta)$; then the posterior p.d.f or p.f. of $\theta$ is 
			\begin{alignat}{4}
				\xi(\theta\mid \mathbf{x}) = \frac{f(x_1\mid\theta)\dots f(x_n\mid \theta)\xi(\theta)}{g_n(\mathbf{x})}
			\end{alignat}
			where $g_n$ is the marginal joint density of $\set{X_i}$. 
		\end{theorem}

		\begin{definition}[Prior Distribution]\index{prior distribution}\index{distribution!prior}
			Suppose that one has a statistical model with parameter $\theta$, If one treats $\theta$ as random then the distribution that one assigns to $\theta$ before observing the other random variables of interest is called its \emph{prior distribution}. We shall commonly use the symbol $\xi(\theta)$ to denote the prior density\index{prior density}\index{density!prior}. 	
		\end{definition}
		
		\begin{definition}[Posterior Distribution]\index{posterior distribution}\index{distribution!posterior}
			Consider a statistical inference problem with parameter $\theta$ and random variables $\set{X_i}_{i=1}^n$ to be observed. The conditional distribution of $\theta$ given $\set{X_i}_{i=1}^n$ is called the \emph{posterior distribution} of $\theta$. The density of $\theta$ given $\set{X_i}_{i=1}^n$ is called the \emph{posterior density} of $\theta$ and is denoted $\xi(\theta\mid x_1, \dots, x_n)$. 
		\end{definition}
		
		\begin{theorem}
			Suppose that $\set{X_i}_{i=1}^n$ form a random sample from a distribution for which the density is $f(x\mid \theta)$. Suppose that the value of the parameter $\theta$ is unknown and the prior density of $\theta$ is $\xi(\theta)$; then the posterior density of $\theta$ is 
			\begin{alignat}{4}
				\xi(\theta\mid \mathbf{x})=\frac{f(x_1\mid\theta)\dots f(x_n\mid\theta)\xi(\theta)}{g_n(\mathbf{x})}
			\end{alignat}
			for $\theta\in\Omega$, where $g_n$ is the marginal joint density of $X_1, \dots, X_n$. 
		\end{theorem}
